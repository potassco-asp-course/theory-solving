  \item<3-> \alert<6,9>{\clingcon} produces infinitely many solutions:
    \begin{itemize}
    \item $a$ is true  and $x$ takes all values greater than 7
    \item $a$ is false and $x$ takes all possible integer values
    \end{itemize}
  \item<4-> \alert<6,9>{\flingo} yields infinitely many solutions:
    \begin{itemize}
    \item $a$ is true and $x$ takes all values greater than 7
    \item $a$ is false and $x$ is undefined
    \end{itemize}
  \item<5-> \alert<7>{\clingodl} obtains only two solutions:
    \begin{itemize}
    \item $a$ is true  and $x$ is $8$ and
    \item $a$ is false and $x$ is undefined
    \end{itemize}
